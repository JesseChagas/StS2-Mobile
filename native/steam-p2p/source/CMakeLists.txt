cmake_minimum_required(VERSION 3.24)
project(sts2_gns C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_STATIC_LIB ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ENABLE_ICE ON CACHE BOOL "" FORCE)
set(USE_STEAMWEBRTC OFF CACHE BOOL "" FORCE)
if(NOT GNS_SOURCE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/GameNetworkingSockets/CMakeLists.txt")
 set(GNS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/GameNetworkingSockets")
endif()
if(NOT GNS_SOURCE)
 include(FetchContent)
 FetchContent_Declare(gns GIT_REPOSITORY https://github.com/ValveSoftware/GameNetworkingSockets.git GIT_TAG d534c19aa760df3fb75fd20db13ba1932b8a5463 GIT_SUBMODULES "")
 FetchContent_MakeAvailable(gns)
 set(GNS_SOURCE "${gns_SOURCE_DIR}")
else()
 add_subdirectory("${GNS_SOURCE}" "${CMAKE_BINARY_DIR}/gns")
endif()
if(ANDROID)
 file(READ "${GNS_SOURCE}/src/steamnetworkingsockets/clientlib/steamnetworkingsockets_socketthread.cpp" GNS_SOCKET_SOURCE)
 if(GNS_SOCKET_SOURCE MATCHES "IsPosix\\(\\) && !IsPlaystation\\(\\) && !IsAndroid\\(\\) && !IsNintendoSwitch\\(\\)")
  message(FATAL_ERROR "Apply source/apply-android-patch.py to GNS_SOURCE before building Android; upstream network enumeration excludes Android.")
 endif()
endif()
target_include_directories(GameNetworkingSockets_s PRIVATE "${GNS_SOURCE}/src")
target_sources(GameNetworkingSockets_s PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/sts2_gns.cpp")
add_library(sts2_gns SHARED empty.cpp)
target_link_libraries(sts2_gns PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,GameNetworkingSockets_s>")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR ANDROID)
 target_link_options(sts2_gns PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
endif()
if(MINGW)
 target_link_options(sts2_gns PRIVATE -static-libgcc -static-libstdc++)
endif()
